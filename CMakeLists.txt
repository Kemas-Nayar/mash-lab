cmake_minimum_required(VERSION 3.20)
project(lunyx_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_compile_definitions(BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
add_executable(controller src/controller.cpp)
target_link_libraries(controller PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
add_executable(agent src/agent.cpp)
target_link_libraries(agent PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(agent PRIVATE -Wall -Wextra -Wpedantic)
